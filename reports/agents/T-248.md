### 4.1 binflow-ci-smoke（commit 烟测）

```groovy
// binflow-ci-smoke - T-248 commit smoke gate (target: ~10 min on 2C).
// Source of truth: bare repo /home/lzw/binflow.git (pushed from macOS main).
// Trigger: SCM poll every minute (bare-repo push visible <= 1 min).
pipeline {
  agent any
  options {
    timestamps()
    timeout(time: 25, unit: 'MINUTES')
    buildDiscarder(logRotator(numToKeepStr: '50'))
  }
  triggers { pollSCM('* * * * *') }
  environment {
    GOPATH      = '/var/jenkins_home/go'
    GOCACHE     = '/var/jenkins_home/.cache/go-build'
    GOPROXY     = 'https://goproxy.cn,direct'
    GOTOOLCHAIN = 'local'
    PATH        = "/usr/local/go/bin:/usr/local/ci-bin:${env.PATH}"
  }
  stages {
    stage('checkout') {
      steps {
        checkout([$class: 'GitSCM',
                  branches: [[name: '*/main']],
                  userRemoteConfigs: [[url: '/home/lzw/binflow.git']],
                  extensions: [[$class: 'CleanBeforeCheckout']]])
        sh 'git log -1 --oneline && echo CHECKOUT_OK'
      }
    }
    stage('go build ./...') {
      steps {
        sh '''#!/bin/bash
set -e
go version
echo "BUILD_START $(date +%s)"
CGO_ENABLED=0 go build ./...
echo "BUILD_END $(date +%s)"
'''
      }
    }
    stage('golangci-lint v2.12.2 (storage/metadata/httpapi/adapter)') {
      steps {
        sh '''#!/bin/bash
set -e
echo "LINT_START $(date +%s)"
golangci-lint run ./internal/storage/... ./internal/metadata/... ./internal/httpapi/... ./internal/adapter/...
echo "LINT_END $(date +%s)"
'''
      }
    }
    stage('go test smoke set (storage+metadata)') {
      steps {
        sh '''#!/bin/bash
set -e
echo "TEST_START $(date +%s)"
CGO_ENABLED=1 go test -count=1 ./internal/storage/... ./internal/metadata/...
echo "TEST_END $(date +%s)"
'''
      }
    }
  }
  post {
    always {
      cleanWs(notFailBuild: true)
    }
  }
}
```

### 4.3 binflow-release（tag 发布，参数化 TAG）

```groovy
// binflow-release - T-248 tag release, parameterized TAG.
// goreleaser-style 6-platform archives (make release VER=$TAG) + linux/amd64
// docker image pushed into BinFlow docker-local (dogfood: BinFlow hosts itself).
pipeline {
  agent any
  options {
    timestamps()
    timeout(time: 90, unit: 'MINUTES')
    buildDiscarder(logRotator(numToKeepStr: '20'))
  }
  parameters {
    string(name: 'TAG', defaultValue: 'v1.0.0-t248', description: 'Release version (feeds make release VER=<TAG>)')
  }
  environment {
    GOPATH      = '/var/jenkins_home/go'
    GOCACHE     = '/var/jenkins_home/.cache/go-build'
    GOPROXY     = 'https://goproxy.cn,direct'
    GOTOOLCHAIN = 'local'
    npm_config_cache = '/var/jenkins_home/.npm'
    REG         = '172.16.58.129:8080'
    IMG         = "${REG}/docker-local/binflow"
    RELEASE_TAG = "${params.TAG ?: 'v1.0.0-t248'}"
    PATH        = "/usr/local/go/bin:/usr/local/ci-bin:${env.PATH}"
  }
  stages {
    stage('checkout') {
      steps {
        checkout([$class: 'GitSCM',
                  branches: [[name: '*/main']],
                  userRemoteConfigs: [[url: '/home/lzw/binflow.git']],
                  extensions: [[$class: 'CleanBeforeCheckout']]])
        sh 'git log -1 --oneline && echo CHECKOUT_OK'
      }
    }
    stage('console chain (embed real SPA)') {
      steps {
        sh '''#!/bin/bash
set -e
echo "CONSOLE_START $(date +%s)"
make console
echo "CONSOLE_END $(date +%s)"
'''
      }
    }
    stage('archives: 6 platforms + checksums (make release)') {
      steps {
        sh '''#!/bin/bash
set -e
echo "RELEASE_START $(date +%s)"
make release VER="$RELEASE_TAG"
echo "RELEASE_END $(date +%s)"
ls -la dist/ | sed -n "1,20p"
'''
      }
    }
    stage('docker image (linux/amd64, prebuilt binary)') {
      steps {
        sh '''#!/bin/bash
set -e
ARCH=dist/binflow_${RELEASE_TAG}_linux_amd64.tar.gz
test -f "$ARCH" || { echo "archive missing: $ARCH"; ls -la dist; exit 1; }
rm -rf docker-ci && mkdir docker-ci
tar -xzf "$ARCH" -C docker-ci/
BIN="docker-ci/binflow_${RELEASE_TAG}_linux_amd64/binflow-server"
test -x "$BIN" || { echo "binary missing: $BIN"; find docker-ci -maxdepth 2 -type f; exit 1; }
mv "$BIN" docker-ci/binflow-server
cat > docker-ci/Dockerfile <<'DOCKERFILE'
FROM alpine:3
RUN sed -i "s|dl-cdn.alpinelinux.org|mirrors.tuna.tsinghua.edu.cn|g" /etc/apk/repositories \
 && apk add --no-cache ca-certificates tzdata \
 && addgroup -S -g 10001 binflow \
 && adduser -S -D -H -u 10001 -G binflow binflow \
 && mkdir -p /var/lib/binflow /etc/binflow \
 && chown -R binflow:binflow /var/lib/binflow /etc/binflow
ENV BINFLOW_DATA_DIR=/var/lib/binflow
COPY --chown=binflow:binflow binflow-server /usr/local/bin/binflow-server
USER binflow:binflow
EXPOSE 8080
VOLUME /var/lib/binflow
HEALTHCHECK --interval=10s --timeout=3s --start-period=5s --retries=5 \
  CMD wget -q -O /dev/null http://127.0.0.1:8080/readyz || exit 1
ENTRYPOINT ["/usr/local/bin/binflow-server"]
CMD ["serve"]
DOCKERFILE
docker build -t "$IMG:$RELEASE_TAG" -t "$IMG:latest" docker-ci/
docker image inspect "$IMG:$RELEASE_TAG" --format "image {{.Id}} size {{.Size}}"
'''
      }
    }
    stage('push to BinFlow docker-local (dogfood)') {
      steps {
        withCredentials([usernamePassword(credentialsId: 'binflow-ci-bot', usernameVariable: 'BF_USER', passwordVariable: 'BF_PASS')]) {
          sh '''#!/bin/bash
set -e
echo "$BF_PASS" | docker login -u "$BF_USER" --password-stdin "$REG"
docker push "$IMG:$RELEASE_TAG"
docker push "$IMG:latest"
docker logout "$REG" || true
'''
        }
      }
    }
    stage('dogfood verify (pull from BinFlow + run + readyz)') {
      steps {
        sh '''#!/bin/bash
set -e
docker rmi "$IMG:$RELEASE_TAG" 2>/dev/null || true
docker pull "$IMG:$RELEASE_TAG"
docker rm -f t248-dogfood 2>/dev/null || true
docker run -d --name t248-dogfood -p 18081:8080 "$IMG:$RELEASE_TAG"
ok=0
for i in $(seq 1 30); do
  if curl -sf http://172.16.58.129:18081/readyz >/dev/null 2>&1; then ok=1; break; fi
  sleep 2
done
[ "$ok" -eq 1 ] || { docker logs t248-dogfood; exit 1; }
echo "readyz: $(curl -s http://172.16.58.129:18081/readyz)"
echo "version: $(curl -s http://172.16.58.129:18081/binflow/api/system/version)"
docker logs t248-dogfood 2>&1 | head -5 || true
docker rm -f t248-dogfood
'''
        withCredentials([usernamePassword(credentialsId: 'binflow-ci-bot', usernameVariable: 'BF_USER', passwordVariable: 'BF_PASS')]) {
          sh '''#!/bin/bash
set -e
curl -sf -u "$BF_USER":"$BF_PASS" "http://$REG/v2/docker-local/binflow/tags/list"
echo
'''
        }
      }
    }
  }
  post {
    always {
      archiveArtifacts artifacts: 'dist/binflow_*.tar.gz,dist/binflow_*.zip,dist/binflow_*checksums.txt', allowEmptyArchive: true, fingerprint: true
      cleanWs(notFailBuild: true)
    }
  }
}
```

## 8. 踩坑（后续 agent 复用）

1. **新版 Jenkins logRotator 参数名**：`numToKeep` → `numToKeepStr`（2.568 编译期直接拒）。`cleanWs` 的 `notDeleteUntrackedPatterns` 同类不存在——别画蛇添足。
2. **REST 更新 job config 必须 `Content-Type: application/xml; charset=utf-8`**：不带 charset 时 Stapler 按 ISO-8859-1 解析 body，groovy 里注释的 em-dash（E2 80 94）第二字节 0x80 被当非法 XML 控制符 → 500 `SAXParseException`。createItem 与 update 路径行为还不一致，统一带上最稳。
3. **镜像里覆盖 PATH 会踢掉 Temurin**：`ENV PATH=/usr/local/go/bin:...` 写全时漏了 `/opt/java/openjdk/bin`，jenkins.sh 落到 Debian openjdk（缺 libharfbuzz）→ BootFailure `libfontmanager.so` UnsatisfiedLinkError 重启循环。追加工具目录时务必保留原 PATH 前缀。
4. **git 插件拒本地路径 checkout**：`ALLOW_LOCAL_CHECKOUT` 系统属性需进 JAVA_OPTS（bare repo 本地挂载方案的必要条件）。
5. **goreleaser v2 dist 布局**：只留 `<id>_<os>_<arch>` 构建目录 + tar.gz 归档，没有 Makefile offline-bundle 注释暗示的 `dist/binflow_<VER>_<os>_<arch>/` 解包目录——取 linux 二进制要从 tar.gz 解（release job 已如此）。
6. **容器 DNS 间歇 NXDOMAIN**：默认网关 DNS 172.16.58.2 对 goproxy.cn 时好时坏（手动解析成功、go build 批量解析失败）。容器 `--dns 223.5.5.5 --dns 119.29.29.29` 解决；另用宿主预灌模块缓存兜底。
7. **agent 容器内探宿主发布端口**：docker run -p 由宿主 daemon 发布，agent 里 `curl 127.0.0.1:<port>` 打的是容器自身——必须用宿主 IP（T-247 §6.4 同族坑，dogfood 探活时又踩一次）。
8. **crumb + cookie jar** 仍是 REST POST 前置（T-247 §6.5 延续）；ssh 远端 heredoc 的重定向要写在引号内（本地 `> /home/lzw/...` 路径不存在会静默废掉整条命令）。
