--- [348-398] start()（无参=前台运行模式）---
start() {
    databaseChecks
    export JF_ARTIFACTORY_PID=${JF_PRODUCT_HOME}/var/work/run/${ARTIFACTORY_NAME}.pid
    ...
    if [ -z "$@" ]; then
        startupActions
        for serviceScript in ${JF_RT_SERVICE_DEPENDENT_SERVICES} ${JF_RT_SERVICE_NON_DEP_SERVICES}; do
            local action="start"
            if [ "${serviceScript}" == "${accessScript}" ]; then
                action=""       # access 以前台（& 后台挂起）方式启动
            fi
            performActionOnScript "${serviceScript}" "${action}"
        done
        waitForDefaultTLSCertsGeneration
        setCatalinaOpts
        if $(isConsoleLogDisabled >/dev/null 2>&1); then
            $TOMCAT_HOME/bin/catalina.sh run
        else
            $TOMCAT_HOME/bin/catalina.sh run > >(tee >(redirectServiceLogsToFile)) 2>&1
        fi
    ...
}

stop() {
    CATALINA_MGNT_PORT=8015

    done
}
